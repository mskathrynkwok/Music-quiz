/* ========================================================= GAME ENGINE ========================================================= */ let currentForm = 1; let currentTopic = ""; let currentQuestion = 0; let score = 0; let answered = false; /* ---------- SCREEN NAVIGATION ---------- */ function showScreen(screenId) { document.querySelectorAll(".screen").forEach(screen => { screen.classList.remove("active"); }); const screen = document.getElementById(screenId); if (screen) { screen.classList.add("active"); } window.scrollTo({ top: 0, behavior: "smooth" }); } /* ---------- SELECT FORM ---------- */ function selectForm(formNumber) { currentForm = formNumber; const topics = QUESTION_BANK[formNumber]; if (!topics) { alert("This Form is not available yet."); return; } document.getElementById("topics-title").textContent = `Form ${formNumber}`; document.getElementById("topics-subtitle").textContent = formNumber === 1 ? "Foundation & Musical Exploration / 基礎與音樂探索" : formNumber === 2 ? "Expression & Musical Structure / 音樂表達與結構" : "Analysis & Musical Thinking / 音樂分析與思考"; renderTopics(); showScreen("topics"); } /* ---------- RENDER TOPICS ---------- */ function renderTopics() { const grid = document.getElementById("topic-grid"); grid.innerHTML = ""; const topics = QUESTION_BANK[currentForm]; if (!topics) { grid.innerHTML = `

No topics available

暫時沒有可用課題。

`; return; } const topicNames = Object.keys(topics); topicNames.forEach((topicName, index) => { const questions = topics[topicName]; const card = document.createElement("div"); card.className = "card topic-card"; card.onclick = function() { startQuiz(topicName); }; card.innerHTML = `
Topic ${index + 1}

${topicName}

${questions.length} questions · ${questions.length} 題
`; grid.appendChild(card); }); } /* ---------- START QUIZ ---------- */ function startQuiz(topicName) { const questions = QUESTION_BANK[currentForm][topicName]; if (!questions || questions.length === 0) { alert("No questions available for this topic."); return; } currentTopic = topicName; currentQuestion = 0; score = 0; answered = false; document.getElementById("quiz-form").textContent = `Form ${currentForm}`; document.getElementById("quiz-topic").textContent = topicName; showScreen("quiz"); renderQuestion(); } /* ---------- RENDER QUESTION ---------- */ function renderQuestion() { const questions = QUESTION_BANK[currentForm][currentTopic]; const question = questions[currentQuestion]; if (!question) { showResult(); return; } answered = false; document.getElementById("question-number").textContent = `Question ${currentQuestion + 1} / ${questions.length} · 第 ${currentQuestion + 1} / ${questions.length} 題`; document.getElementById("quiz-counter").textContent = `${currentQuestion + 1} / ${questions.length}`; document.getElementById("question-en").textContent = question.q[0]; document.getElementById("question-zh").textContent = question.q[1]; const progress = ((currentQuestion) / questions.length) * 100; document.getElementById("progress").style.width = `${progress}%`; const optionsContainer = document.getElementById("options"); optionsContainer.innerHTML = ""; question.options.forEach((option, index) => { const button = document.createElement("button"); button.className = "option"; button.innerHTML = ` ${option[0]} ${option[1]} `; button.onclick = function() { chooseAnswer(index); }; optionsContainer.appendChild(button); }); document.getElementById("feedback").style.display = "none"; document.getElementById("next-btn").style.display = "none"; } /* ---------- ANSWER ---------- */ function chooseAnswer(selectedIndex) { if (answered) return; answered = true; const questions = QUESTION_BANK[currentForm][currentTopic]; const question = questions[currentQuestion]; const optionButtons = document.querySelectorAll(".option"); optionButtons.forEach(button => { button.disabled = true; }); optionButtons[question.answer].classList.add("correct"); if (selectedIndex === question.answer) { score++; } else { optionButtons[selectedIndex].classList.add("wrong"); } const feedback = document.getElementById("feedback"); const correctOption = question.options[question.answer]; const isCorrect = selectedIndex === question.answer; feedback.innerHTML = ` ${isCorrect ? "✓ Correct! / 正確！" : "✗ Not quite. / 未能答對。"}
Correct answer / 正確答案： ${correctOption[0]} / ${correctOption[1]}
${question.ex[0]}
${question.ex[1]}
`; feedback.style.display = "block"; const nextButton = document.getElementById("next-btn"); nextButton.style.display = "inline-block"; if (currentQuestion === questions.length - 1) { nextButton.textContent = "See Results / 查看結果"; } else { nextButton.textContent = "Next / 下一題"; } } /* ---------- NEXT QUESTION ---------- */ function nextQuestion() { const questions = QUESTION_BANK[currentForm][currentTopic]; if (!answered) return; if (currentQuestion >= questions.length - 1) { showResult(); return; } currentQuestion++; renderQuestion(); } /* ---------- RESULT ---------- */ function showResult() { const questions = QUESTION_BANK[currentForm][currentTopic]; const total = questions.length; const percentage = Math.round((score / total) * 100); document.getElementById("score").textContent = `${score} / ${total}`; let message; if (percentage >= 80) { message = `Excellent! / 表現出色！
You demonstrated a strong understanding of this topic.
你對這個課題已經有很好的掌握。`; } else if (percentage >= 60) { message = `Good work! / 做得不錯！
Keep practising and connecting the concepts.
繼續練習，並嘗試把不同音樂概念連繫起來。`; } else { message = `Keep exploring! / 繼續探索！
Review the questions and try the topic again.
可以重溫題目，再挑戰一次。`; } document.getElementById("result-message").innerHTML = message; document.getElementById("progress").style.width = "100%"; showScreen("result"); } /* ---------- RESTART ---------- */ function restartQuiz() { currentQuestion = 0; score = 0; answered = false; showScreen("quiz"); renderQuestion(); } /* ---------- LANGUAGE BUTTON ---------- */ let englishFirst = true; function toggleLanguage() { englishFirst = !englishFirst; const button = document.querySelector(".lang-btn"); button.textContent = englishFirst ? "中文 / English" : "English / 中文"; }
